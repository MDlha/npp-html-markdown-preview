cmake_minimum_required(VERSION 3.24)
project(npp-html-markdown-preview VERSION 0.8 LANGUAGES C CXX RC)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/${PROJECT_VERSION}")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK "${CMAKE_SOURCE_DIR}/vendor/webview2/sdk/build/native")
if(NOT EXISTS "${SDK}/include/WebView2.h")
  message(FATAL_ERROR "Missing pinned WebView2 SDK; see BUILDING.md")
endif()
add_library(markdown STATIC src/Markdown.cpp vendor/md4c/src/md4c.c vendor/md4c/src/md4c-html.c vendor/md4c/src/entity.c)
target_include_directories(markdown PUBLIC src PRIVATE vendor/md4c/src)
target_compile_options(markdown PRIVATE /utf-8)
add_library(preview_window STATIC src/PreviewWindow.cpp src/DocumentWindow.cpp src/RuntimeSupport.cpp)
target_include_directories(preview_window PUBLIC src PRIVATE "${SDK}/include")
target_compile_definitions(preview_window PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(preview_window PRIVATE /W4 /utf-8 /guard:cf)
target_link_libraries(preview_window PUBLIC "${SDK}/x64/WebView2LoaderStatic.lib" ole32 user32 shell32 version shlwapi urlmon advapi32)
add_library(npp-html-markdown-preview SHARED src/Plugin.cpp src/EditorSnapshot.cpp src/SourcePosition.cpp src/Version.rc)
set_target_properties(npp-html-markdown-preview PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/out/${PROJECT_VERSION}")
target_include_directories(npp-html-markdown-preview PRIVATE vendor/plugintemplate/src)
target_link_libraries(npp-html-markdown-preview PRIVATE preview_window markdown)
target_compile_options(npp-html-markdown-preview PRIVATE /W4 /utf-8 /guard:cf)
target_link_options(npp-html-markdown-preview PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/PluginAbiCheck.cpp")
  add_executable(PreviewProbe WIN32 src/Probe.cpp)
  add_executable(HtmlPreview WIN32 src/HtmlPreview.cpp)
  target_link_libraries(HtmlPreview PRIVATE preview_window markdown comdlg32)
  target_compile_options(HtmlPreview PRIVATE /W4 /utf-8 /guard:cf)
  target_link_libraries(PreviewProbe PRIVATE preview_window)
  target_compile_options(PreviewProbe PRIVATE /W4 /utf-8 /guard:cf)
  enable_testing()
  add_test(NAME probe_self_check COMMAND PreviewProbe --self-check)
  add_executable(PluginAbiCheck tests/PluginAbiCheck.cpp)
  target_include_directories(PluginAbiCheck PRIVATE src vendor/plugintemplate/src)
  target_compile_definitions(PluginAbiCheck PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(PluginAbiCheck PRIVATE /W4 /utf-8)
  add_test(NAME plugin_abi COMMAND PluginAbiCheck $<TARGET_FILE:npp-html-markdown-preview>)
  add_executable(EditorSnapshotCheck tests/EditorSnapshotCheck.cpp src/EditorSnapshot.cpp src/SourcePosition.cpp)
  target_include_directories(EditorSnapshotCheck PRIVATE src vendor/plugintemplate/src)
  target_compile_definitions(EditorSnapshotCheck PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(EditorSnapshotCheck PRIVATE /W4 /utf-8)
  target_link_libraries(EditorSnapshotCheck PRIVATE user32)
  add_test(NAME editor_snapshot COMMAND EditorSnapshotCheck)
  add_executable(MarkdownCheck tests/MarkdownCheck.cpp)
  target_link_libraries(MarkdownCheck PRIVATE markdown)
  target_compile_options(MarkdownCheck PRIVATE /utf-8)
  add_test(NAME markdown_render COMMAND MarkdownCheck)
  add_executable(SourcePositionCheck tests/SourcePositionCheck.cpp src/SourcePosition.cpp)
  target_include_directories(SourcePositionCheck PRIVATE src)
  target_compile_options(SourcePositionCheck PRIVATE /utf-8)
  add_test(NAME source_position COMMAND SourcePositionCheck)
  get_filename_component(MSVC_TOOL_DIRECTORY "${CMAKE_CXX_COMPILER}" DIRECTORY)
  find_program(PREVIEW_DUMPBIN NAMES dumpbin.exe HINTS "${MSVC_TOOL_DIRECTORY}" REQUIRED)
  add_test(NAME plugin_no_external_crt COMMAND ${CMAKE_COMMAND} "-DDUMPBIN=${PREVIEW_DUMPBIN}" "-DBINARY=$<TARGET_FILE:npp-html-markdown-preview>" -P "${CMAKE_SOURCE_DIR}/tests/CheckRuntime.cmake")
  add_test(NAME standalone_no_external_crt COMMAND ${CMAKE_COMMAND} "-DDUMPBIN=${PREVIEW_DUMPBIN}" "-DBINARY=$<TARGET_FILE:HtmlPreview>" -P "${CMAKE_SOURCE_DIR}/tests/CheckRuntime.cmake")
endif()
